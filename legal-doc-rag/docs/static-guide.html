def get_safe_upload_path(upload_dir, tenant_id, filename):
    safe_tenant = re.sub(r'[^a-zA-Z0-9_-]', '', tenant_id)   # ①清洗 tenant_id
    safe_filename = sanitize_filename(filename)              # ②清洗文件名
    upload_dir = os.path.join(upload_dir, safe_tenant)
    os.makedirs(upload_dir, exist_ok=True)
    file_path = os.path.join(upload_dir, safe_filename)
    if not is_safe_path(upload_dir, file_path):              # ③终极校验
        raise ValueError(f"Unsafe file path detected: {filename}")
    return file_path

def is_safe_path(base_dir, target_path) -> bool:
    base = Path(base_dir).resolve()        # 解析符号链接/../ 到真实绝对路径
    target = Path(target_path).resolve()
    return target.is_relative_to(base)     # target 必须仍在 base 之下

class MultimodalPipeline:
    def __init__(self):
        self._splitter = RecursiveCharacterTextSplitter(
            chunk_size=500, chunk_overlap=50,
            separators=["\n\n","\n","。","；","，"," "])   # 中文按句/段断
    def _process_pdf(self, pdf_path):
        pages = extract_pdf_pages(pdf_path)
        for page in pages:
            # ① 图片 OCR 出文字  ② 图片用视觉模型生成描述
            combined = page_text
            if caption_texts: combined += "\n" + "\n".join(caption_texts)
            if ocr_texts:     combined += "\n[图片文字]\n" + "\n".join(ocr_texts)
            for st in self._splitter.split_text(combined):  # ③ 切 500 字块
                chunks.append(MultimodalChunk(text=st, page_number=page["page_number"]))
        return chunks

关键代码 · 向量化按 index 排序（步骤 5，易错点）

class DirectEmbed:
    def embed_documents(self, texts):
        resp = requests.post(f"{self.base_url}/embeddings",
            json={"model": self.model, "input": texts}, timeout=120)
        data = resp.json()
        # 批量 embedding 返回顺序未必与输入一致，必须按 index 排回来！
        return [item["embedding"] for item in sorted(data["data"], key=lambda x: x["index"])]

关键代码 · 混合检索 + RRF 融合（步骤 4）

def retrieve(self, query, top_k=None):
    k = top_k or self.k
    dense  = self._dense_search(query)    # ① Chroma 向量相似度
    sparse = self._sparse_search(query)   # ② BM25 关键词
    es     = self._elasticsearch_search(query)  # ③ ES（可选）
    bge    = self._sparse_search_bge(query)      # ④ BGE-M3 稀疏(SPLADE)
    fused  = self._rrf_fuse(dense, sparse, es, bge)  # ⑤ RRF 融合
    if self.reranker and self.reranker.available:
        fused = self.reranker.rerank(query, fused, k)  # ⑥ BGE 精排
    return fused

def _rrf_fuse(self, dense_results, sparse_results, es_results=None, bge_sparse_results=None):
    doc_map = {}
    for rank,(doc,_) in enumerate(dense_results):
        key = doc.page_content[:200]
        if key not in doc_map: doc_map[key]=doc; doc_map[key].metadata["rrf_score"]=0.0
        doc_map[key].metadata["rrf_score"] += self.dense_weight/(self.rrf_k+rank+1)
    # BM25 / ES / BGE-M3 稀疏 同法累加（只按排名给分，不依赖分数绝对值）
    return sorted(doc_map.values(), key=lambda d:d.metadata["rrf_score"], reverse=True)

RRF 本质：不比较向量分和 BM25 分（量纲不同），只按"排名"加分 1/(60+rank)，三路都靠前的自然总分最高。

关键代码 · BGE 交叉编码器精排（步骤 5）

class Reranker:
    def __init__(self, model_name="BAAI/bge-reranker-base"):
        self.model=None; self.available=False
        try:
            from sentence_transformers import CrossEncoder
            self.model = CrossEncoder(model_name, device="cpu")
            self.available = True
        except Exception as e:
            logger.warning("Reranker unavailable (skip): {}", e)  # 失败跳过，不阻塞
    def rerank(self, query, documents, top_k=5):
        if not self.available or not documents: return documents[:top_k]
        pairs = [[query, d.page_content[:512]] for d in documents]
        scores = self.model.predict(pairs)   # 把 (query,doc) 拼一起交叉编码
        scored = sorted(zip(documents, scores), key=lambda x:x[1], reverse=True)
        return [d for d,_ in scored[:top_k]]

关键代码 · 引用追踪（步骤 6，可溯源实现）

class CitationTracker:
    def format_context(self) -> str:
        parts = []
        for i, src in enumerate(self._sources):
            marker = f"[{i+1}]"          # 对应 Prompt 要求的"用 [N] 引用"
            ref = src.filename
            if src.page_number: ref += f" 第{src.page_number}页"
            parts.append(f"{marker} {ref}\n{src.content}")
        return "\n\n".join(parts)

4 · 两个最常考点

①租户隔离 = 根目录 + tenant_id 拼子目录；安全靠 get_safe_upload_path()(middleware.py:114) 清洗 + is_safe_path()(:102) 防 ../ 穿越。
②hybrid + reranker 而非纯语义：法律术语要精确匹配避免漏检；向量找语义相似，BM25 保精确，BGE 精排提精度。记住这两点，架构就吃透了。
