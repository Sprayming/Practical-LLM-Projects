class SQLGenerator:
    def generate_sql(self, question: str, schema: str) -> str:
        prompt = (
            "你是一个数据分析助手。基于下表结构，把用户问题转成一条安全的 SQL。\n"
            f"表结构:\n{schema}\n问题:{question}\n只输出 SQL，不要解释。"
        )
        resp = self.llm.chat(prompt, temperature=0)   # temperature=0 保证稳定
        sql = self._extract_sql(resp)
        return sql

    def describe_tables(self) -> str:
        # 从数据库元数据拼出建表语句/字段说明，作为 LLM 上下文
        return self.schema_parser.describe()

关键代码 · 安全双防线（步骤 2，重点）

def sanitize_input(text: str) -> str:
    # 去控制字符、压缩空白、限长
    text = re.sub(r"[\x00-\x1f\x7f]", "", text)
    text = re.sub(r"\s+", " ", text).strip()
    return text[:500]

def validate_question(text: str) -> bool:
    # 禁止纯 SQL/命令注入式提问；只接受自然语言分析意图
    if re.search(r"(?i)\b(drop|delete|update|insert|truncate|alter)\b", text):
        return False
    if ";" in text and ("--" in text or "/*" in text):
        return False
    return True

关键代码 · SQL 校验 + 执行（步骤 4）

def validate_sql(sql: str) -> bool:
    parsed = sqlparse.parse(sql)[0]
    # 只允许 SELECT；禁止多语句、写操作、系统库
    if not parsed.get_type() == "SELECT": return False
    if ";" in sql.strip().rstrip(";"): return False
    forbidden = ("information_schema","mysql","pg_","sqlite_master")
    if any(f in sql.lower() for f in forbidden): return False
    return True

def run_sql(sql: str):
    if not validate_sql(sql): raise PermissionError("不安全的 SQL 被拦截")
    with get_connection() as conn:        # db/connection.py 连接池
        return pd.read_sql(sql, conn)

关键代码 · 共享记忆（Agent 间通信）

class SharedMemory:
    def __init__(self): self._store: dict[str, list[Message]] = {}
    def publish(self, agent: str, msg: Message):
        self._store.setdefault(agent, []).append(msg)
    def query(self, topic: str) -> list[Message]:
        return [m for msgs in self._store.values() for m in msgs if m.topic == topic]

class Message:
    def __init__(self, topic: str, payload: dict, sender: str):
        self.topic = topic; self.payload = payload; self.sender = sender
